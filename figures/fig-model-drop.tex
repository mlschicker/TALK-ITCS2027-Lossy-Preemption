% Lossy preemption, alternative view: an interruption deletes progress.
% The job runs during [0,2], is interrupted, and resumes during [3.5,6.5].
% At the interruption its progress is Y = 2; drawn for gamma = 1/2:
%   only gamma*Y = 1 is retained, (1-gamma)*Y = 1 is lost (red arrow),
%   and after resuming the job is back at Y at time 4.5 (red triangle = regained).
% Same end result as fig-model.tex: progress 4 at time 6.5.
\begin{tikzpicture}[x=0.85cm,y=0.65cm]
  \renewcommand{\barh}{0.9}
  % --- processing intervals (light background) ---------------------------
  \fill[jobA!7] (0,0)   rectangle (2,5.4);
  \fill[jobA!7] (3.5,0) rectangle (6.5,5.4);
  % --- axes ---------------------------------------------------------------
  \draw[axis] (0,0) -- (7.4,0) node[below right,lbl]{time};
  \draw[axis] (0,0) -- (0,5.9) node[above,lbl]{progress};
  \foreach \t in {0,...,7} \draw[tick] (\t,0) -- ++(0,-0.15) node[below,lbl]{$\t$};
  \foreach \y in {1,...,5} \draw[tick] (0,\y) -- ++(-0.1,0) node[left,lbl]{$\y$};
  % --- old level Y and retained level gamma*Y ------------------------------
  \draw[guide] (0,2) -- (6.5,2) node[right,lbl]{$Y$};
  \draw[guide] (0,1) -- (6.5,1) node[right,lbl]{$\gamma Y$};
  % --- lost progress being regained: nothing new below the old level Y -----
  \fill[alertred!20] (3.5,1) -- (4.5,2) -- (3.5,2) -- cycle;
  % --- the two classical extremes (grey) ---------------------------------
  \draw[line width=0.9pt, black!45, dashed]         (2,2) -- (3.5,2) -- (6.5,5) node[right,lbl]{$\gamma = 1$};
  \draw[line width=0.9pt, black!45, densely dotted] (2,2) -- (2,0) -- (3.5,0) -- (6.5,3) node[right,lbl]{$\gamma = 0$};
  % --- progress for gamma = 1/2: drop at the interruption ------------------
  \draw[line width=1.4pt, jobA] (0,0) -- (2,2);
  \draw[line width=1.4pt, jobA] (2,1) -- (3.5,1) -- (6.5,4) node[right,lbl]{$\gamma = \tfrac12$};
  \draw[->, line width=1pt, alertred] (2,2) -- (2,1.08);
  \node[font=\scriptsize, alertred] at (2.8,1.5) {$(1-\gamma)Y$};
  % --- the same schedule as a Gantt strip: no waiting ----------------------
  \job[-2.6]{jobA}{0}{2}{job $j$}
  \job[-2.6]{other}{2}{3.5}{others}
  \job[-2.6]{jobA}{3.5}{6.5}{job $j$}
\end{tikzpicture}
