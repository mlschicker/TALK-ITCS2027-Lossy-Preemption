% Parallel machines: random assignment, then randomized b-scaling per machine.
% Jobs are the small squares on the left; machine rows are y = 2, 1, 0.

  % --- job pool (colour = job) -------------------------------------------

  \foreach \y in {2.3,1.3,0.3}
